import java.util.*;

public class Main {
    public static void main(String args[]){
        Scanner sc = new Scanner(System.in);
        
        int n = sc.nextInt();
        int[] arr = new int[n];
        
        for(int i=0; i<n; i++){
            arr[i] = sc.nextInt();
        }
        
        int maxSoFar = arr[n-1];
        
        for(int i = n - 2; i>=0; i--){
            int currentElement= arr[i];
            arr[i] = maxSoFar;
            if(currentElement > maxSoFar){
                maxSoFar = currentElement;
           }
        }
        for(int i = 0; i< n; i++){
            System.out.print(arr[i]);
            if(i<n-1){
                System.out.print(" ");
            }
        }
        System.out.println();
        
        
    }
}
